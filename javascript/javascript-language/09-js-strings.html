2.2. Common string methods

const s = "  Hello, World  ";

s.length;                    // 17
s.trim();                    // "Hello, World"
s.toUpperCase();             // "  HELLO, WORLD  "
s.toLowerCase();
s.slice(0, 5);               // "  Hel"
s.substring(2, 7);           // "Hello"
s.includes("World");         // true
s.startsWith("  H");         // true
s.endsWith("d  ");
s.indexOf("o");              // 8
s.lastIndexOf("o");
s.replace("World", "JS");    // "  Hello, JS  "
s.replaceAll("l", "L");      // replace all (ES2021)
s.split(", ");               // ["  Hello", "World  "]
s.concat("!");               // prefer + or template literals
"abc".repeat(3);             // "abcabcabc"
"5".padStart(3, "0");        // "005"

2.3. Tagged templates

function highlight(strings, ...values) {
  return strings.reduce((acc, str, i) => {
    return acc + str + (values[i] ? `<b>${values[i]}</b>` : "");
  }, "");
}

const user = "Ada";
highlight`User: ${user}`;  // "User: <b>Ada</b>"
  • File 22 covers tagged templates in ES6+ context.

2.3.1. String.raw

  • String.raw is a built-in tag that keeps backslashes raw (no escape processing) — handy for paths and regex text:
String.raw`C:\Users\name\file`;   // "C:\\Users\\name\\file" as written
`C:\Users\name`;                   // escapes may surprise

String.raw`line1\nline2`;          // backslash-n stays two chars, not a newline
const path = String.raw`\d+`;      // useful when building RegExp strings

L2 Intermediate

2.4. Unicode, UTF-16 Surrogates & Grapheme Clusters

  • JavaScript strings are internally sequences of 16-bit code units (UTF-16). Characters in the Basic Multilingual Plane (BMP, code points ≤ 0xFFFF) occupy 1 code unit. Supplementary characters (emojis, rare ideographs > 0xFFFF) require a surrogate pair (two 16-bit code units: high surrogate + low surrogate).
"😀".length;                    // 2 — two 16-bit code units!
[..."😀"].length;                // 1 — spread operator iterates full code points
"😀".codePointAt(0);            // 128512 (full Unicode scalar value)
"😀".charCodeAt(0);             // 55357 (0xD83D, only the high surrogate!)
String.fromCodePoint(128512);   // "😀"

// Compound Emojis & Grapheme Clusters (e.g., family or flag with ZWJ):
const family = "👨‍👩‍👧‍👦";
console.log(family.length);      // 11 (UTF-16 code units)
console.log([...family].length); // 7 (Unicode code points joined by Zero-Width Joiners)

// For true user-perceived character length, use Intl.Segmenter (ES2022):
const segmenter = new Intl.Segmenter("en", { granularity: "grapheme" });
const graphemes = [...segmenter.segment(family)];
console.log(graphemes.length);   // 1 (1 visible grapheme cluster!)
  • Always use for...of or [...str] for code points, and Intl.Segmenter when splitting user-visible international text.

2.5. String immutability

  • Strings are immutable — methods return new strings:
let s = "hello";
s.toUpperCase();  // "HELLO"
console.log(s);     // still "hello"

2.6. Locale, normalize, well-formed

"a".localeCompare("b");           // negative — a before b
"ä".localeCompare("z", "de");

"é".normalize("NFC") === "\u00e9";  // composed
"e\u0301".normalize("NFC");         // "é" — same after NFC

"\uD800".isWellFormed();            // false — lone surrogate
"hi".toWellFormed();                // replaces unpaired surrogates

C. Common questions

L1 L2

L1 Fundamental

C.1. Are strings mutable? ▾ Answer

No — methods return new strings.

L1 Fundamental

C.2. What is the difference between slice and substring? ▾ Answer

Both extract substrings. slice accepts negative indices; substring swaps negative to 0.

L2 Intermediate

C.3. Why does "😀".length equal 2? ▾ Answer

.length counts UTF-16 code units; emoji use surrogate pairs. Use spread or for...of for code points.

C.4. How do I search or replace with a pattern? ▾ Answer

File 10 — RegExp, flags, match/matchAll/replace.

C.5. How do I check if a string is a palindrome? ▾ Answer

Normalize (lowercase, strip non-letters if needed), then compare to its reverse:

function isPalindrome(s) {
  const t = s.toLowerCase().replace(/[^a-z0-9]/g, "");
  return t === [...t].reverse().join("");
}
isPalindrome("A man, a plan, a canal: Panama");  // true

P.2. Core tasks

  1. normalizeTitle — Given a messy title string, trim it, collapse internal spaces, and Title-Case each word (split / map / join). ▾ Solution 
    function normalizeTitle(rawTitle) {
  return rawTitle
    .trim()
    .replace(/\s+/g, " ")
    .split(" ")
    .map(word => word.charAt(0).toUpperCase() + word.slice(1).toLowerCase())
    .join(" ");
}

console.log(normalizeTitle("   jaVaScRipt   lanGUage   noTes  "));
// "Javascript Language Notes"
  2. maskEmail — For "ada@example.com", return something like "a***@example.com" (keep first char of local part). ▾ Solution 
    function maskEmail(email) {
  const [local, domain] = email.split("@");
  if (!local || !domain) return email;
  const maskedLocal = local[0] + "*".repeat(Math.max(1, local.length - 1));
  return `${maskedLocal}@${domain}`;
}

console.log(maskEmail("ada@example.com"));     // "a**@example.com"
console.log(maskEmail("bharat@domain.org"));   // "b*****@domain.org"
  3. slugify — Lowercase, replace spaces with -, strip non-alphanumeric (except -). ▾ Solution 
    function slugify(text) {
  return text
    .toLowerCase()
    .trim()
    .replace(/[^a-z0-9\s-]/g, "") // remove non-alphanumeric chars except space/dash
    .replace(/\s+/g, "-")         // replace spaces with single hyphen
    .replace(/-+/g, "-");          // collapse multiple hyphens
}

console.log(slugify("  Modern JavaScript: 101 Guide & Tips!  "));
// "modern-javascript-101-guide-tips"
  4. padReceipt — Use padStart / padEnd to align labels and prices in three logged lines. ▾ Solution 
    function formatReceiptItem(item, price) {
  const formattedItem = item.padEnd(16, ".");
  const formattedPrice = ("$" + price.toFixed(2)).padStart(8, ".");
  return `${formattedItem}${formattedPrice}`;
}

console.log(formatReceiptItem("Coffee", 3.5));
console.log(formatReceiptItem("Croissant", 4.25));
console.log(formatReceiptItem("Chocolate Cake", 6.0));
// Output:
// Coffee.............$3.50
// Croissant..........$4.25
// Chocolate Cake.....$6.00

L2 Intermediate

P.3. Challenge

  1. isPalindrome — Implement the interview version (ignore case and non-alphanumerics). Test with "A man, a plan, a canal: Panama". ▾ Solution 
    function isPalindrome(str) {
  // Strip non-alphanumeric characters and lowercase
  const clean = str.toLowerCase().replace(/[^a-z0-9]/g, "");
  const reversed = clean.split("").reverse().join("");
  return clean === reversed;
}

console.log(isPalindrome("A man, a plan, a canal: Panama")); // true
console.log(isPalindrome("race a car"));                     // false
console.log(isPalindrome("Was it a car or a cat I saw?"));   // true
  2. Code points — Log "😀".length, then count characters with [...str] or for...of. Comment why they differ. ▾ Solution 
    const emoji = "😀";

// UTF-16 code units length:
console.log("emoji.length:", emoji.length); // 2 (surrogate pair)

// Actual Unicode character count:
const codePoints = Array.from(emoji);
console.log("Array.from(emoji).length:", codePoints.length); // 1
console.log("Spread length:", [...emoji].length);            // 1
