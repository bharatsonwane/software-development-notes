4.1. Array.prototype.map

function map(arr, fn) {
  const result = [];
  for (let i = 0; i < arr.length; i++) {
    if (i in arr) result[i] = fn(arr[i], i, arr);
  }
  return result;
}

4.2. Array.prototype.filter

function filter(arr, fn) {
  const result = [];
  for (let i = 0; i < arr.length; i++) {
    if (i in arr && fn(arr[i], i, arr)) result.push(arr[i]);
  }
  return result;
}

4.3. Array.prototype.reduce

function reduce(arr, fn, initial) {
  let acc = initial;
  let start = 0;
  if (initial === undefined) {
    if (arr.length === 0) throw new TypeError("Reduce of empty array");
    acc = arr[0];
    start = 1;
  }
  for (let i = start; i < arr.length; i++) {
    if (i in arr) acc = fn(acc, arr[i], i, arr);
  }
  return acc;
}

4.4. call and apply

  • Naive (uses native apply) — fine for demos, not a real polyfill:
function call(fn, thisArg, ...args) {
  return fn.apply(thisArg, args);
}
// apply same with args array
  • Interview sketch without apply — temporarily attach fn as a property on thisArg, invoke it, then clean up:
function myCall(fn, thisArg, ...args) {
  const ctx = thisArg == null ? globalThis : Object(thisArg);
  const key = Symbol("call");
  ctx[key] = fn;
  const result = ctx[key](...args);
  delete ctx[key];
  return result;
}

4.5. flatten array

function flatten(arr, depth = 1) {
  return depth > 0
    ? arr.reduce((acc, v) => acc.concat(Array.isArray(v) ? flatten(v, depth - 1) : v), [])
    : arr.slice();
}

4.7. Promise (minimal)

class MyPromise {
  constructor(executor) {
    this.state = "pending";
    this.value = undefined;
    this.handlers = [];
    const resolve = (val) => {
      if (this.state !== "pending") return;
      this.state = "fulfilled";
      this.value = val;
      this.handlers.forEach((h) => h.onFulfilled?.(val));
    };
    const reject = (err) => {
      if (this.state !== "pending") return;
      this.state = "rejected";
      this.value = err;
      this.handlers.forEach((h) => h.onRejected?.(err));
    };
    try { executor(resolve, reject); } catch (e) { reject(e); }
  }
  then(onFulfilled, onRejected) {
    return new MyPromise((resolve, reject) => {
      const handle = () => {
        try {
          const next = this.state === "fulfilled"
            ? onFulfilled?.(this.value) ?? this.value
            : onRejected?.(this.value);
          resolve(next);
        } catch (e) { reject(e); }
      };
      if (this.state === "pending") this.handlers.push({ onFulfilled: handle, onRejected: handle });
      else handle();
    });
  }
}

4.8. Promise.all and Promise.race

function promiseAll(promises) {
  return new Promise((resolve, reject) => {
    const results = [];
    let done = 0;
    promises.forEach((p, i) => {
      Promise.resolve(p).then(
        (val) => { results[i] = val; if (++done === promises.length) resolve(results); },
        reject
      );
    });
  });
}

function chunk(arr, size) {
  return Array.from({ length: Math.ceil(arr.length / size) }, (_, i) =>

5.6. Deep clone

function deepClone(val) {
  if (val === null || typeof val !== "object") return val;
  if (Array.isArray(val)) return val.map(deepClone);
  return Object.fromEntries(
    Object.entries(val).map(([k, v]) => [k, deepClone(v)])
  );
}
// Production: structuredClone(val) — File 02, 07

5.7. Retry logic

async function retry(fn, times = 3, delayMs = 500) {
  for (let i = 0; i < times; i++) {
    try { return await fn(); }
    catch (e) { if (i === times - 1) throw e; await sleep(delayMs); }
  }
}

5.8. Object flattening

function flattenObject(obj, prefix = "", out = {}) {
  for (const [k, v] of Object.entries(obj)) {
    const key = prefix ? `${prefix}.${k}` : k;
    if (v && typeof v === "object" && !Array.isArray(v)) flattenObject(v, key, out);
    else out[key] = v;
  }
  return out;
}

5.9. Nested object traversal

function walk(obj, fn, path = "") {
  for (const [k, v] of Object.entries(obj)) {
    const p = path ? `${path}.${k}` : k;
    if (v && typeof v === "object" && !Array.isArray(v)) walk(v, fn, p);
    else fn(p, v);
  }
}

L3 Advanced

5.10. Promise pool (concurrency limit)

async function promisePool(tasks, limit) {
  const results = [];
  const executing = [];
  for (const [i, task] of tasks.entries()) {
    const p = Promise.resolve().then(task).then((r) => { results[i] = r; });
    results[i] = p;
    if (tasks.length >= limit) {
      executing.push(p);
      if (executing.length >= limit) await Promise.race(executing);
      executing.splice(executing.findIndex((x) => x === p), 1);
    }
  }
  await Promise.all(results);
  return results;
}

6. Security basics

L1 L2

L1 Fundamental

6.1. XSS (Cross-Site Scripting)

  • Injected script runs in victim's browser. Prevent: escape output, textContent not innerHTML, CSP, sanitize HTML (DOMPurify).

6.2. Sanitization

  • Never trust user HTML. Validate on server; encode on output.

L2 Intermediate

6.3. CSRF (Cross-Site Request Forgery)

  • Tricks browser into sending authenticated request. Prevent: CSRF tokens, SameSite cookies.

6.4. CORS

  • Browser blocks cross-origin responses unless server sends Access-Control-Allow-Origin. Preflight for non-simple requests.

6.5. CSP (Content Security Policy)

  • HTTP header restricting script/style sources — mitigates XSS.

6.6. eval and Function

  • eval(string) and new Function(string) run arbitrary code. Never feed them user input — XSS and injection risk. Prefer JSON.parse, templating libraries, or explicit parsers.

6.7. Secure storage

  • localStorage / sessionStorage are readable by any script on the page — if XSS exists, tokens there are stolen.
  • Prefer httpOnly, Secure, SameSite cookies for session credentials when the server can set them. Treat client storage as non-secret UI state.

7.3. De-optimization

Function.prototype.myBind = function(context, ...boundArgs) {
  const targetFn = this;

  3. debounce + throttle — From File 23, rewrite from memory. ▾ Solution 
    function debounce(fn, delay) {
  let timer;
  return function(...args) {
    clearTimeout(timer);
    timer = setTimeout(() => fn.apply(this, args), delay);
  };
}

function throttle(fn, limit) {
  let inThrottle = false;
  return function(...args) {
    if (!inThrottle) {
      fn.apply(this, args);
      inThrottle = true;
      setTimeout(() => (inThrottle = false), limit);
    }
  };
}

  1. promise polyfill sketch — Minimal thenable that supports then/catch happy path (educational, not A+). ▾ Solution 
    class MyPromise {
  constructor(executor) {
    this.state = "pending";
    this.value = undefined;
    this.handlers = [];

    const resolve = (val) => {
      if (this.state !== "pending") return;
      this.state = "fulfilled";
      this.value = val;
      this.handlers.forEach((h) => h(val));
    };

    try {
      executor(resolve);
    } catch (err) {
      // rejection handling in full implementation
      throw err;
    }
  }

  then(onFulfilled) {
    if (this.state === "fulfilled") {
      onFulfilled(this.value);
    } else {
      this.handlers.push(onFulfilled);
    }
    return this;
  }
}

  2. LRU cache — Small Map-based LRU with get/set and capacity. ▾ Solution 
    class LRUCache {
  constructor(capacity) {
    this.capacity = capacity;
    this.cache = new Map(); // Maintains insertion order
  }

  get(key) {
    if (!this.cache.has(key)) return -1;
    const value = this.cache.get(key);
    // Refresh position by re-inserting
    this.cache.delete(key);
    this.cache.set(key, value);
    return value;
  }

  put(key, value) {
    if (this.cache.has(key)) {
      this.cache.delete(key);
    } else if (this.cache.size >= this.capacity) {
      // Evict least recently used (first key in map iterator)
      const lruKey = this.cache.keys().next().value;
      this.cache.delete(lruKey);
    }
    this.cache.set(key, value);
  }
}

const lru = new LRUCache(2);
lru.put("a", 1);
lru.put("b", 2);
console.log("Get a:", lru.get("a")); // 1 (makes 'a' most recently used)
lru.put("c", 3);                     // Evicts 'b'
console.log("Get b:", lru.get("b")); // -1 (evicted!)
console.log("Get c:", lru.get("c")); // 3
