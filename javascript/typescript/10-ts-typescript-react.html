2. Typing components and props

L1 L2

L1 Fundamental

2.1. Function components

type ButtonProps = {
  label: string;
  onClick: () => void;
  disabled?: boolean;
};

function Button({ label, onClick, disabled = false }: ButtonProps) {
  return (
    <button type="button" onClick={onClick} disabled={disabled}>
      {label}
    </button>
  );
}
  • Or with interface:
interface CardProps {
  title: string;
  children: React.ReactNode;
}

function Card({ title, children }: CardProps) {
  return (
    <section>
      <h2>{title}</h2>
      {children}
    </section>
  );
}

2.2. Typing children

type Props = {
  children: React.ReactNode; // string, number, element, fragment, null, array, etc.
};
  • Stricter alternatives: React.ReactElement (single element), React.ReactNode (most common).

L2 Intermediate

2.3. React.FC / React.FunctionComponent

  • Older style — optional children was implicit. Modern codebases prefer explicit props:
// Avoid unless team standard
const OldButton: React.FC<ButtonProps> = ({ label, onClick }) => (
  <button onClick={onClick}>{label}</button>
);

2.4. Extending HTML element props

type InputProps = React.InputHTMLAttributes<HTMLInputElement> & {
  label: string;
};

function LabeledInput({ label, ...rest }: InputProps) {
  return (
    <label>
      {label}
      <input {...rest} />
    </label>
  );
}
  • Use ComponentPropsWithoutRef<"button"> or ComponentProps<typeof SomeComponent> to inherit props.

L3 Advanced

2.5. Polymorphic Components (The as Prop)

  • Build reusable UI elements that can render as a <button>, <a>, or custom component while retaining complete type safety for HTML attributes:
type PolymorphicButtonProps<E extends React.ElementType = "button"> = {
  as?: E;
  children: React.ReactNode;
} & React.ComponentPropsWithoutRef<E>;

function PolymorphicButton<E extends React.ElementType = "button">({
  as,
  children,
  ...rest
}: PolymorphicButtonProps<E>) {
  const Component = as || "button";
  return <Component {...rest}>{children}</Component>;
}

// Usage:
<PolymorphicButton onClick={() => console.log("clicked")}>Native Button</PolymorphicButton>;
<PolymorphicButton as="a" href="https://example.com" target="_blank">Link Styled As Button</PolymorphicButton>;

const [count, setCount] = useState(0); // number

type ListProps = {
  items: string[];
  onSelect: (item: string, index: number) => void;
};

L2 Intermediate

3.3. Event handlers

  • React events are synthetic — use types from @types/react:
function SearchForm() {
  const [query, setQuery] = useState("");

  const handleChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    setQuery(e.target.value);
  };

  const handleSubmit = (e: React.FormEvent<HTMLFormElement>) => {
    e.preventDefault();
    console.log(query);
  };

  return (
    <form onSubmit={handleSubmit}>
      <input value={query} onChange={handleChange} />
    </form>
  );
}
  • Common event types:
EventType
Input changeReact.ChangeEvent<HTMLInputElement>
Button clickReact.MouseEvent<HTMLButtonElement>
Form submitReact.FormEvent<HTMLFormElement>
KeyboardReact.KeyboardEvent<HTMLElement>

4. Typing hooks and context

L1 L2

L1 Fundamental

4.1. Custom hooks

  • Return type is often inferred; annotate for public hooks:
function useCounter(initial = 0) {
  const [count, setCount] = useState(initial);
  const increment = () => setCount((c) => c + 1);
  return { count, increment, setCount };
}

type UseCounterReturn = ReturnType<typeof useCounter>;
  • Generic custom hook:
function useLocalStorage<T>(key: string, initial: T) {
  const [value, setValue] = useState<T>(() => {
    const raw = localStorage.getItem(key);
    return raw ? (JSON.parse(raw) as T) : initial;
  });

  useEffect(() => {
    localStorage.setItem(key, JSON.stringify(value));
  }, [key, value]);

  return [value, setValue] as const;
}

L2 Intermediate

4.2. useReducer

type State = { count: number };
type Action =
  | { type: "increment" }
  | { type: "decrement" }
  | { type: "reset" };

function reducer(state: State, action: Action): State {
  switch (action.type) {
    case "increment":
      return { count: state.count + 1 };
    case "decrement":
      return { count: state.count - 1 };
    case "reset":
      return { count: 0 };
  }
}

function Counter() {
  const [state, dispatch] = useReducer(reducer, { count: 0 });
  // ...
}

4.3. Context

type Theme = "light" | "dark";

type ThemeContextValue = {
  theme: Theme;
  toggle: () => void;
};

const ThemeContext = createContext<ThemeContextValue | null>(null);

function useTheme() {
  const ctx = useContext(ThemeContext);
  if (!ctx) {
    throw new Error("useTheme must be used within ThemeProvider");
  }
  return ctx;
}

function ThemeProvider({ children }: { children: React.ReactNode }) {
  const [theme, setTheme] = useState<Theme>("light");
  const toggle = () => setTheme((t) => (t === "light" ? "dark" : "light"));

  return (
    <ThemeContext.Provider value={{ theme, toggle }}>
      {children}
    </ThemeContext.Provider>
  );
}

L3 Advanced

4.4. Typing useRef (DOM Ref vs Mutable Variable Ref)

  • useRef has two distinct TypeScript overloads based on its initial value:
// Overload 1: DOM Element Reference (initial value is null)
// Returns: React.RefObject<HTMLInputElement> (current is read-only)
const inputRef = useRef<HTMLInputElement>(null);

function focusInput() {
  inputRef.current?.focus(); // inputRef.current is HTMLInputElement | null
}

// Overload 2: Mutable Instance Variable (e.g. interval ID, persistent cache)
// Returns: React.MutableRefObject<number> (current is fully mutable)
const timerId = useRef<number>(0);

function startTimer() {
  timerId.current = window.setInterval(() => console.log("tick"), 1000); // OK to mutate!
}

5. Third-party types (@types/*)

L1 L2

L1 Fundamental

  • Many JS libraries ship types. Others need DefinitelyTyped packages:
npm install --save-dev @types/react @types/react-dom
npm install --save-dev @types/node
npm install --save-dev @types/lodash

5.1. Library includes its own types

  • Modern packages embed .d.ts:
// package.json of a typed library
{
  "name": "some-lib",
  "types": "dist/index.d.ts"
}
  • No @types/some-lib needed.

L2 Intermediate

5.2. No types available

  • Create a declaration file:
// types/untyped-lib.d.ts
declare module "untyped-lib" {
  export function doSomething(input: string): number;
}
  • Or use // @ts-expect-error temporarily — fix with proper types or wrapper.

5.3. ref typing

const inputRef = useRef<HTMLInputElement>(null);

useEffect(() => {
  inputRef.current?.focus();
}, []);

return <input ref={inputRef} />;

C. Common questions

L1 L2

L1 Fundamental

C.1. Should I use interface or type for props? ▾ Answer

Either. type needed for unions (Props | DisabledProps). Teams often pick one and stay consistent.

C.2. What is React.ReactNode vs React.ReactElement? ▾ Answer

ReactNode is anything renderable. ReactElement is a single JSX element — stricter.

C.3. Do I need @types/react with React 18+? ▾ Answer

Types are still published via @types/react — install as dev dependency.

C.4. What about propTypes? ▾ Answer

TypeScript replaces PropTypes for TS projects. PropTypes optional for JS interop.

L2 Intermediate

C.5. Why useState<string[]>([]) instead of useState([])? ▾ Answer

Empty array infers never[] — explicit generic sets element type.

C.6. How do I type forwardRef? ▾ Answer

const Input = forwardRef<HTMLInputElement, InputProps>((props, ref) => ...).

C.7. How to type generic table/list components? ▾ Answer

function List<T>({ items, render }: { items: T[]; render: (item: T) => React.ReactNode }).

C.8. Strict null and optional props? ▾ Answer

disabled?: boolean allows undefined. With exactOptionalPropertyTypes, omitting vs passing undefined differs.

type LoadState = "idle" | "loading" | "success" | "error";

function StatusDisplay() {

  const [status, setStatus] = useState<LoadState>("idle");

  1. children/ReactNode. ▾ Solution

  2. useToggle hook types. ▾ Solution 
    import { useState } from "react";

// Explicit return tuple type: [boolean, () => void]
function useToggle(initial: boolean = false): [boolean, () => void] {
  const [value, setValue] = useState<boolean>(initial);
  const toggle = () => setValue((v) => !v);
  return [value, toggle];
}

export default useToggle;
  3. typed context. ▾ Solution 
    import React, { createContext, useContext, useState, ReactNode } from "react";

interface AuthContextType {
  user: string | null;
  login: (name: string) => void;
  logout: () => void;
}

const AuthContext = createContext<AuthContextType | undefined>(undefined);

export function AuthProvider({ children }: { children: ReactNode }) {
  const [user, setUser] = useState<string | null>(null);

  return (
    <AuthContext.Provider value={{ user, login: setUser, logout: () => setUser(null) }}>
      {children}
    </AuthContext.Provider>
  );
}

export function useAuth(): AuthContextType {
  const context = useContext(AuthContext);
  if (!context) throw new Error("useAuth must be used within an AuthProvider");
  return context;
}

P.3. Challenge

  1. Discriminated button props. ▾ Solution 
    import React from "react";

// Discriminated union for Button props:
// If href is provided, it acts as an <a> link; otherwise it is a <button>.
type ButtonAsButton = {
  href?: undefined;
} & React.ButtonHTMLAttributes<HTMLButtonElement>;

type ButtonAsLink = {
  href: string;
} & React.AnchorHTMLAttributes<HTMLAnchorElement>;

type ButtonProps = ButtonAsButton | ButtonAsLink;

function DynamicButton(props: ButtonProps) {
  if (props.href !== undefined) {
    return <a {...props} className="btn-link">{props.children}</a>;
  }
  return <button {...props} className="btn-button">{props.children}</button>;
}

export default DynamicButton;
  2. Form handlers without any. ▾ Solution 
    import React, { useState } from "react";

function FormWithProperTypes() {
  const [email, setEmail] = useState("");

  const handleInputChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    setEmail(e.target.value);
  };

  const handleFormSubmit = (e: React.FormEvent<HTMLFormElement>) => {
    e.preventDefault();
    console.log("Submitting email:", email);
  };

  return (
    <form onSubmit={handleFormSubmit}>
      <input type="email" value={email} onChange={handleInputChange} />
      <button type="submit">Submit</button>
    </form>
  );
}

export default FormWithProperTypes;
