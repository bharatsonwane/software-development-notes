def grade_student(score):
    if not (0 <= score <= 100):
        return "Invalid Score"
    elif score >= 90:
        return "A+ (Outstanding)"
    elif score >= 80:
        return "A (Excellent)"
    elif score >= 70:
        return "B (Good)"

    else:
        return "F (Fail)"

# Sum of first N odd numbers using for and range:
n = 5
odd_sum = 0
for i in range(1, 2 * n, 2):
    odd_sum += i
print(f"Sum of first {n} odd numbers:", odd_sum)

# Fibonacci sequence using while loop:
a, b = 0, 1
fib_series = []
while len(fib_series) < 8:
    fib_series.append(a)
    a, b = b, a + b
print("First 8 Fibonacci numbers:", fib_series)

# Pyramid of stars:
rows = 5
for i in range(1, rows + 1):
    spaces = " " * (rows - i)
    stars = "*" * (2 * i - 1)
    print(f"{spaces}{stars}")

# Number triangle:
for i in range(1, rows + 1):
    for j in range(1, i + 1):
        print(j, end=" ")
    print()

    # Mini Project: Prime Number Analyzer & Statistics Generator
def analyze_primes(limit):
    primes = []
    for candidate in range(2, limit + 1):
        is_prime = True
        for divisor in range(2, int(candidate ** 0.5) + 1):
            if candidate % divisor == 0:
                is_prime = False
                break
        if is_prime:
            primes.append(candidate)
    
    print(f"Primes up to {limit}: {primes}")
    print(f"Total count: {len(primes)}, Largest prime: {primes[-1] if primes else None}")

analyze_primes(30)
