    # Summary of String Creation and Indexing:
# 1. Strings in Python are immutable sequences of Unicode code points enclosed in quotes.
# 2. Positive indexing starts from 0 (left-to-right); negative indexing starts from -1 (right-to-left).
# 3. Item assignment (s[0] = 'x') raises TypeError because strings cannot be mutated in-place.
  2. Create a tiny example related to Slice operator and print the result. ▾ Solution 
    # Slice operator syntax: s[begin:end:step]
message = "PythonMastery"
print("Full:", message[:])
print("First 6 chars:", message[:6])     # 'Python'
print("Last 7 chars:", message[6:])      # 'Mastery'
print("Step by 2:", message[::2])        # 'PtoMser'
print("Reversed string:", message[::-1]) # 'yretsaMnohtyP'

    # Task 2: Slice operator demo
data = "ABCDEFGHIJ"
print("Slice [2:7]:", data[2:7])       # 'CDEFG'
print("Slice [:5]:", data[:5])         # 'ABCDE'
print("Slice [5:]:", data[5:])         # 'FGHIJ'
print("Slice [1:9:2]:", data[1:9:2])   # 'BDFH'
print("Slice [::-1]:", data[::-1])     # 'JIHGFEDCBA' (Reversal)
print("Negative slice [-5:-2]:", data[-5:-2]) # 'FGH'

# 1. Palindrome check:
def is_palindrome(word):
    clean = "".join(c.lower() for c in word if c.isalnum())
    return clean == clean[::-1]

print("is_palindrome('A man, a plan, a canal: Panama'):", is_palindrome("A man, a plan, a canal: Panama"))

# 2. Word frequency count:
quote = "code clean code fast debug code"
freq = {}

    freq[w] = freq.get(w, 0) + 1
print("Word frequency:", freq)

    # Mini Project: Text Sanitizer & Slug Generator
import re

def create_url_slug(title, max_length=50):
    # Normalize and remove special characters:
    slug = title.strip().lower()
    slug = re.sub(r'[^a-z0-9\s-]', '', slug)
    slug = re.sub(r'[\s-]+', '-', slug)
    slug = slug[:max_length].rstrip('-')
    return slug

titles = [
    "10 Essential Python Tips for 2026!",
    "Building Fast, Scalable REST APIs with Python & FastAPI",
    "Understanding Memory Interning & Garbage Collection..."
]

for t in titles:
    print(f"Title: {t}\nSlug:  https://example.com/posts/{create_url_slug(t)}\n")
  2. Edge cases — List 3 edge cases or mistakes beginners make; demonstrate at least one safely. ▾ Solution 
    # String Pitfalls & Edge Cases:
# 1. Strings are immutable:
s = "hello"
# s[0] = 'H' # TypeError: 'str' object does not support item assignment
s = "H" + s[1:] # Correct way

# 2. find() returns -1 on not found; index() raises ValueError:
print("'python'.find('z'):", 'python'.find('z')) # -1

# 3. Accidental repeated concatenation in loops is O(N^2); prefer str.join():
words = ["python", "is", "awesome"]
result = " ".join(words) # O(N) efficient
