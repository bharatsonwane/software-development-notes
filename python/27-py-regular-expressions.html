    # Character classes, predefined classes, and quantifiers:
import re

text = "Invoice #INV-2026-4982 paid on 13-09-2026"
# Match pattern: INV followed by 4 digits, then 4 digits
pattern = r"INV-\d{4}-\d{4}"
match = re.search(pattern, text)
if match:
    print("Found Invoice:", match.group())

  1. Task 1: Introduction to regular expressions — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 1: Introduction to regular expressions
import re

pattern = r"\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Z|a-z]{2,}\b"
sample_text = "Contact support at info@example.com or admin@corp.org for help."

emails = re.findall(pattern, sample_text)
print("Extracted email addresses:", emails)
  2. Task 2: Character classes, predefined classes, quantifiers — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 2: Character classes, predefined classes, quantifiers
import re

# Predefined classes: \d (digit), \w (alphanumeric), \s (whitespace)
# Quantifiers: * (0+), + (1+), ? (0 or 1), {min,max}
pattern = r"\w+@\w+\.\w+"
text = "Users: user1@mail.com and dev_team@git.io"
print("Matches:", re.findall(pattern, text))

# Custom character class:
vowels_or_digits = r"[aeiou0-9]"
print("Vowels and digits in 'Python3':", re.findall(vowels_or_digits, "Python3", re.I))
  3. Task 3: Important functions of re module — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 3: Important functions of re module
import re

text = "apple, orange; banana: grape"
# re.split with multiple delimiters:
fruits = re.split(r"[,;:]\s*", text)
print("re.split:", fruits)

# re.sub for replacement:
censored = re.sub(r"\d{4}", "****", "My pin is 1234 and card code is 5678")
print("re.sub:", censored)

# re.finditer for positions:
for match in re.finditer(r"\b\w{5}\b", "seven apple tiger zebra"):
    print(f"5-letter word: '{match.group()}' at [{match.start()}:{match.end()}]")
  4. Task 4: Practice applications — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 4: Practice applications (Phone & URL validator)
import re

def validate_phone(number):
    # Matches (123) 456-7890 or 123-456-7890
    regex = r"^(\(\d{3}\)|\d{3})[- ]?\d{3}[- ]?\d{4}$"
    return bool(re.match(regex, number))

test_numbers = ["123-456-7890", "(555) 123-4567", "12345", "999-abc-4567"]
for num in test_numbers:
    print(f"'{num}': Valid = {validate_phone(num)}")

    # Mini Project: Log File Analyzer & Metric Extractor
import re

log_entries = [
    '192.168.1.10 - [13/Sep/2026:10:00:01] "GET /api/v1/users HTTP/1.1" 200 452',
    '10.0.0.5 - [13/Sep/2026:10:00:02] "POST /api/v1/login HTTP/1.1" 401 128',
    '172.16.0.2 - [13/Sep/2026:10:00:05] "GET /static/style.css HTTP/1.1" 304 0',
    '192.168.1.10 - [13/Sep/2026:10:00:10] "GET /api/v1/orders HTTP/1.1" 500 230'
]

log_pattern = re.compile(
    r'(?P<ip>[\d\.]+)\s-\s\[(?P<time>[^\]]+)\]\s"(?P<method>\w+)\s(?P<path>[^\s]+)[^"]*"\s(?P<status>\d{3})'
)

print("Parsed Web Server Access Logs:")
for entry in log_entries:
    match = log_pattern.search(entry)
    if match:
        data = match.groupdict()
        print(f"IP: {data['ip']:<14} | Method: {data['method']:<4} | Status: {data['status']} | Route: {data['path']}")
