# Simulating package __init__.py namespace exposure:
class PackageNamespace:
    """Simulates what package __init__.py exposes to callers."""
    VERSION = "2.5.0"
    __all__ = ["connect", "disconnect"]
    
    @staticmethod
    def connect():
        return "Connected to service cluster"

pkg = PackageNamespace()
print(f"Package Version: {pkg.VERSION}")
print("Exposed method:", pkg.connect())

    # Mini Project: Mock Plugin Architecture with Package-style Registry
class PluginRegistry:
    def __init__(self):
        self.plugins = {}
    
    def register(self, name):
        def decorator(cls):
            self.plugins[name] = cls()
            return cls
        return decorator
    
    def run_all(self, data):
        results = {}
        for name, plugin in self.plugins.items():
            results[name] = plugin.execute(data)
        return results

registry = PluginRegistry()

@registry.register("uppercase_transformer")
class UppercasePlugin:
    def execute(self, text):
        return text.upper()

@registry.register("word_counter")
class WordCountPlugin:
    def execute(self, text):
        return len(text.split())

results = registry.run_all("Python modular package architecture")
print("Plugin execution results:", results)
