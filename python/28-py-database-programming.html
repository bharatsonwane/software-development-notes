  2. Task 2: Create/drop/insert (sample) — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 2: Create/drop/insert table operations
import sqlite3

conn = sqlite3.connect(":memory:")
cur = conn.cursor()

# Create table:
cur.execute("""
    CREATE TABLE products (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        name TEXT NOT NULL,
        price REAL NOT NULL
    )
""")

# Parameterized bulk insert (executemany):
items = [
    ("Wireless Keyboard", 49.99),
    ("Gaming Mouse", 29.50),
    ("HDMI Cable", 9.99)
]
cur.executemany("INSERT INTO products (name, price) VALUES (?, ?)", items)
conn.commit()
print(f"Inserted {cur.rowcount} records successfully.")
conn.close()
  3. Task 3: CRUD operations — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 3: Full CRUD operations (Create, Read, Update, Delete)
import sqlite3

conn = sqlite3.connect(":memory:")
cur = conn.cursor()
cur.execute("CREATE TABLE tasks (id INTEGER PRIMARY KEY, title TEXT, done BOOLEAN)")

# 1. Create:
cur.execute("INSERT INTO tasks (title, done) VALUES (?, ?)", ("Buy milk", False))
task_id = cur.lastrowid

# 2. Read:
cur.execute("SELECT * FROM tasks WHERE id = ?", (task_id,))
print("Read:", cur.fetchone())

# 3. Update:
cur.execute("UPDATE tasks SET done = ? WHERE id = ?", (True, task_id))
conn.commit()

# 4. Delete:
cur.execute("DELETE FROM tasks WHERE id = ?", (task_id,))
conn.commit()
print("Tasks count after delete:", cur.execute("SELECT count(*) FROM tasks").fetchone()[0])
conn.close()

  1. Mini project — Combine two topics from this file into one short script related to Database Programming (PDBC). ▾ Solution 
    # Mini Project: In-Memory Transactional Bank Account Ledger
import sqlite3

conn = sqlite3.connect(":memory:")
cur = conn.cursor()

cur.execute("CREATE TABLE accounts (id INTEGER PRIMARY KEY, holder TEXT, balance REAL)")
cur.execute("INSERT INTO accounts (holder, balance) VALUES ('Alice', 1000.0), ('Bob', 500.0)")
conn.commit()

def transfer_funds(from_id, to_id, amount):
    try:
        cur.execute("BEGIN TRANSACTION")
        # Deduct:
        cur.execute("UPDATE accounts SET balance = balance - ? WHERE id = ?", (amount, from_id))
        # Add:
        cur.execute("UPDATE accounts SET balance = balance + ? WHERE id = ?", (amount, to_id))
        conn.commit()
        print(f"Successfully transferred ${amount:.2f} from Account #{from_id} to #{to_id}")
    except Exception as err:
        conn.rollback()
        print(f"Transaction failed, rolled back: {err}")

transfer_funds(1, 2, 200.0)

for row in cur.execute("SELECT id, holder, balance FROM accounts"):
    print(f"  Account #{row[0]} ({row[1]}): ${row[2]:.2f}")
conn.close()
