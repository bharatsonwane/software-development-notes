  2. Task 2: Command-line arguments (sys.argv) — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 2: Command-line arguments with sys.argv
import sys

def parse_cli_args(args):
    program_name = args[0]
    cli_params = args[1:]
    print(f"Running program: {program_name}")
    for idx, arg in enumerate(cli_params, 1):
        print(f"  Param {idx}: {arg}")
    return len(cli_params)

# Simulated execution:
mock_argv = ["app.py", "--port", "8080", "--env", "production"]
total_params = parse_cli_args(mock_argv)
print(f"Processed {total_params} CLI arguments successfully.")

def format_bill(item_name, price_usd, exchange_rate=83.5):
    price_inr = price_usd * exchange_rate
    print("=" * 40)
    print(f"{'RECEIPT':^40}")
    print("=" * 40)
    print(f"Item:       {item_name}")
    print(f"Price USD:  ${price_usd:.2f}")
    print(f"Rate:       1 USD = {exchange_rate:.2f} INR")
    print("-" * 40)
    print(f"Total INR:  ₹{price_inr:,.2f}")
    print("=" * 40)

format_bill("Mechanical Keyboard", 129.99)

# 1. input() always returns str, leading to unintended string concatenation:

print("Mistake (str + str):", a + b)  # '1020' instead of 30!
print("Correct (int + int):", int(a) + int(b))  # 30
