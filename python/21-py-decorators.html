  2. Create a tiny example related to Introduction to decorators and print the result. ▾ Solution 
    # Introduction to decorators:
# A decorator is a function that takes another function, adds behavior, and returns it.
def my_decorator(func):
    def wrapper():
        print("[Before function execution]")
        func()
        print("[After function execution]")
    return wrapper

@my_decorator
def say_hello():
    print("Hello from inner function!")

say_hello()

  2. Task 2: Introduction to decorators — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 2: Introduction to decorators with arguments
from functools import wraps

def log_execution(func):
    @wraps(func)
    def wrapper(*args, **kwargs):
        print(f"[LOG] Calling {func.__name__} with args={args}, kwargs={kwargs}")
        result = func(*args, **kwargs)
        print(f"[LOG] {func.__name__} returned {result}")
        return result
    return wrapper

@log_execution
def add_numbers(a, b):
    return a + b

sum_result = add_numbers(15, 27)
  3. Task 3: Decorator demo patterns — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 3: Decorator demo patterns (timing decorator)
import time
from functools import wraps

def measure_time(func):
    @wraps(func)
    def wrapper(*args, **kwargs):
        start = time.perf_counter()
        result = func(*args, **kwargs)
        elapsed = time.perf_counter() - start
        print(f"{func.__name__} took {elapsed:.6f}s")
        return result
    return wrapper

@measure_time
def compute_sum(n):
    return sum(range(n))

print("Total sum:", compute_sum(100000))
  4. Task 4: Calling with/without decorator; chaining — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 4: Calling with/without decorator; chaining
def bold(func):
    def wrapper(*args, **kwargs):
        return f"<b>{func(*args, **kwargs)}</b>"
    return wrapper

def italic(func):
    def wrapper(*args, **kwargs):
        return f"<i>{func(*args, **kwargs)}</i>"
    return wrapper

# Decorator chaining: applied bottom-to-top:
@bold
@italic
def format_text(text):
    return text

print("Chained output:", format_text("Python Decorators"))
# Equivalent to: bold(italic(format_text))("Python Decorators")

    # Mini Project: Rate Limiter / Cache Memoization Decorator
from functools import wraps

def memoize(func):
    cache = {}
    @wraps(func)
    def wrapper(*args):
        if args in cache:
            print(f"[CACHE HIT] for {args}")
            return cache[args]
        print(f"[COMPUTING] for {args}")
        result = func(*args)
        cache[args] = result
        return result
    return wrapper

@memoize
def fibonacci(n):
    if n < 2:
        return n
    return fibonacci(n - 1) + fibonacci(n - 2)

print("fib(10):", fibonacci(10))
print("fib(10) cached:", fibonacci(10))
