P.2. Core tasks

  1. Task 1: Operator overloading and __str__ / __repr__ — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 1: Operator overloading and __str__ / __repr__
class Vector2D:
    def __init__(self, x, y):
        self.x = x
        self.y = y
    
    def __add__(self, other):
        if isinstance(other, Vector2D):
            return Vector2D(self.x + other.x, self.y + other.y)
        raise TypeError("Operand must be Vector2D")
    
    def __str__(self):
        return f"({self.x}, {self.y})"
    
    def __repr__(self):
        return f"Vector2D(x={self.x}, y={self.y})"

v1 = Vector2D(3, 4)
v2 = Vector2D(1, 2)
v3 = v1 + v2
print("v1 + v2 =", v3)
print("repr(v3) =", repr(v3))
  2. Task 2: Method and constructor overloading patterns — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 2: Method and constructor overloading patterns
class UserProfile:
    def __init__(self, username, email=None, **extra):
        # Emulates constructor overloading
        self.username = username
        self.email = email or f"{username}@default.org"
        self.extra = extra

# Instantiation with 1 arg, 2 args, or kwargs:
u1 = UserProfile("alice")
u2 = UserProfile("bob", "bob@corp.com")
u3 = UserProfile("charlie", role="Admin", verified=True)

print(f"u1: {u1.username}, {u1.email}")
print(f"u2: {u2.username}, {u2.email}")
print(f"u3 extra: {u3.extra}")
  3. Task 3: Method and constructor overriding — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 3: Method and constructor overriding
class PaymentProcessor:
    def process_payment(self, amount):
        print(f"Base processor: initiating ${amount:.2f} transaction")

class StripeProcessor(PaymentProcessor):
    def process_payment(self, amount):
        super().process_payment(amount)
        print(f"Stripe: Tokenized card and charged ${amount:.2f} via Stripe API")

proc = StripeProcessor()
proc.process_payment(99.99)
  4. Task 4: Abstract methods and abstract classes — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 4: Abstract methods and abstract classes (abc module)
from abc import ABC, abstractmethod

class Shape(ABC):
    @abstractmethod
    def area(self):
        """Subclasses must implement area."""
        pass

class Circle(Shape):
    def __init__(self, radius):
        self.radius = radius
    def area(self):
        return 3.14159 * (self.radius ** 2)

c = Circle(5)
print("Circle area:", round(c.area(), 2))

# s = Shape() # TypeError: Can't instantiate abstract class Shape with abstract method area

  1. Mini project — Combine two topics from this file into one short script related to Polymorphism and Encapsulation. ▾ Solution 
    # Mini Project: Polymorphic Notification Dispatcher
from abc import ABC, abstractmethod

class Notifier(ABC):
    @abstractmethod
    def send(self, recipient, message):
        pass

class EmailNotifier(Notifier):
    def send(self, recipient, message):
        return f"[EMAIL to {recipient}] {message}"

class SMSNotifier(Notifier):
    def send(self, recipient, message):
        return f"[SMS to {recipient}] {message}"

class PushNotifier(Notifier):
    def send(self, recipient, message):
        return f"[PUSH to {recipient}] {message}"

# Polymorphic broadcast:
notifiers = [
    (EmailNotifier(), "user@example.com"),
    (SMSNotifier(), "+1-555-0199"),
    (PushNotifier(), "device_token_xyz")
]

for notifier, target in notifiers:
    print(notifier.send(target, "System maintenance scheduled tonight at 11 PM UTC"))
