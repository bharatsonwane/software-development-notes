  3. Task 3: with statement; seek() and tell() — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 3: with statement; seek() and tell()
with open("seek_demo.txt", "w+") as f:
    f.write("0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ")
    print("Position after writing:", f.tell())  # 36
    
    # Reposition pointer to index 10:
    f.seek(10)
    print("Position after seek(10):", f.tell())
    print("Read 5 chars:", f.read(5))  # 'ABCDE'
    print("Position after reading:", f.tell())  # 15
    
    # Seek back to start:
    f.seek(0)
    print("Read first 4 chars:", f.read(4))  # '0123'

import os
os.remove("seek_demo.txt")
  4. Task 4: Binary, CSV, and ZIP files — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 4: Binary, CSV, and ZIP files
import csv
import zipfile
import os

# 1. Writing CSV:
csv_file = "employees.csv"
with open(csv_file, "w", newline="") as f:
    writer = csv.writer(f)
    writer.writerow(["ID", "Name", "Dept"])
    writer.writerow([101, "Alice", "Dev"])
    writer.writerow([102, "Bob", "QA"])

# 2. Reading CSV:
with open(csv_file, "r") as f:
    reader = csv.reader(f)
    print("CSV Contents:")
    for row in reader:
        print(" ", row)

# 3. Creating ZIP archive:
zip_name = "archive.zip"
with zipfile.ZipFile(zip_name, "w") as zipf:
    zipf.write(csv_file)
print(f"Created {zip_name} containing {csv_file}")

# Clean up
os.remove(csv_file)
os.remove(zip_name)

import os

def analyze_and_backup_log(log_filename):
    # Create sample log if not exists:
    with open(log_filename, "w") as f:
        f.write("ERROR 2026-09-13 Database timeout\n")
        f.write("INFO  2026-09-13 Server started\n")
        f.write("ERROR 2026-09-13 Connection refused\n")
    
    # Analyze lines:
    error_count = 0
    with open(log_filename, "r") as f:
        for line in f:
            if "ERROR" in line:
                error_count += 1
    
    print(f"Log analysis for {log_filename}: {error_count} error events detected.")
    os.remove(log_filename)

analyze_and_backup_log("server.log")
