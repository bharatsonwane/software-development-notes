    # Summary of Introduction to Multithreading:
# 1. A thread is a separate flow of execution within the same process sharing address space.
# 2. Python's 'threading' module provides thread creation and synchronization primitives.
# 3. CPython uses the Global Interpreter Lock (GIL), making threads ideal for I/O-bound tasks, while multiprocessing is used for CPU-bound tasks.
  2. Create a tiny example related to Creating and starting threads and print the result. ▾ Solution 
    # Creating and starting threads:
import threading
import time

def print_worker(message):
    time.sleep(0.05)
    print(f"[Thread {threading.current_thread().name}] says: {message}")

t = threading.Thread(target=print_worker, args=("Hello Multithreading!",), name="Worker-1")
t.start()
t.join()
print("Worker thread completed.")

  1. Task 1: Introduction to multithreading — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 1: Introduction to multithreading
import threading
import time

def worker(num):
    print(f"Task {num} started on {threading.current_thread().name}")
    time.sleep(0.02)
    print(f"Task {num} finished")

threads = []
for i in range(3):
    th = threading.Thread(target=worker, args=(i+1,))
    threads.append(th)
    th.start()

for th in threads:
    th.join()
print("All tasks completed concurrently.")
  2. Task 2: Creating and starting threads — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 2: Creating threads by subclassing threading.Thread
import threading
import time

class CustomWorkerThread(threading.Thread):
    def __init__(self, thread_id, duration):
        super().__init__()
        self.thread_id = thread_id
        self.duration = duration
    
    def run(self):
        print(f"Custom thread #{self.thread_id} running...")
        time.sleep(self.duration)
        print(f"Custom thread #{self.thread_id} finished after {self.duration}s")

worker_inst = CustomWorkerThread(thread_id=42, duration=0.03)
worker_inst.start()
worker_inst.join()

import threading

def identify_thread():
    curr = threading.current_thread()
    print(f"Thread Name: {curr.name}")
    print(f"Thread Ident: {curr.ident}")
    print(f"Is Alive: {curr.is_alive()}")
    print(f"Is Daemon: {curr.daemon}")

t = threading.Thread(target=identify_thread, name="Worker-Named-Alpha")
t.start()
t.join()
print("Active thread count:", threading.active_count())
  4. Task 4: Daemon threads — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 4: Daemon threads
import threading
import time

def background_service():
    print("Daemon background monitor started.")
    time.sleep(0.05)
    print("Daemon background monitor completed.")

# Daemon threads terminate automatically when main thread terminates
d_thread = threading.Thread(target=background_service, daemon=True)
print("Is daemon thread:", d_thread.daemon)
d_thread.start()
d_thread.join()

  1. Mini project — Combine two topics from this file into one short script related to Multithreading. ▾ Solution 
    # Mini Project: Thread-Safe Counter with threading.Lock
import threading

class ThreadSafeCounter:
    def __init__(self):
        self.count = 0
        self.lock = threading.Lock()
    
    def increment(self):
        with self.lock:
            # Critical section protected against race conditions
            self.count += 1

counter = ThreadSafeCounter()
threads = []
for _ in range(50):
    t = threading.Thread(target=counter.increment)
    threads.append(t)
    t.start()

for t in threads:
    t.join()

print("Final synchronized counter value:", counter.count)
  2. Edge cases — List 3 edge cases or mistakes beginners make; demonstrate at least one safely. ▾ Solution 
    # Multithreading Pitfalls & Edge Cases:
# 1. Race Conditions: Multiple threads mutating shared data without a Lock leads to corrupted state.

# 2. Deadlock: Two threads waiting on each other's locks. Always acquire locks in the same order.

# 3. CPU-bound concurrency: Threads won't speed up pure CPU number-crunching in CPython due to the GIL; use multiprocessing instead.
