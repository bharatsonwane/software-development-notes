P.3. Challenge

  1. Mini project — Combine two topics from this file into one short script related to Sets. ▾ Solution 
    # Mini Project: User Permissions & Access Control Evaluator
def check_access(user_roles, required_permissions, role_permissions_map):
    # Aggregate all permissions granted by user's roles using set union:
    user_perms = set()
    for role in user_roles:
        user_perms.update(role_permissions_map.get(role, set()))
    
    missing_perms = required_permissions - user_perms
    has_access = len(missing_perms) == 0
    return has_access, missing_perms

role_map = {
    "editor": {"read", "write"},
    "admin": {"read", "write", "delete", "manage_users"},
    "viewer": {"read"}
}

req = {"read", "write", "delete"}
ok, missing = check_access(["editor"], req, role_map)
print(f"Editor has full access: {ok}, Missing: {missing}")

ok_admin, _ = check_access(["admin"], req, role_map)
print(f"Admin has full access: {ok_admin}")

    # Set Pitfalls & Edge Cases:
# 1. Sets cannot contain mutable items (unhashable):
# s = {[1, 2], [3, 4]}  # TypeError: unhashable type: 'list'
s = {(1, 2), (3, 4)}    # Tuples are fine!

# 2. Sets are unordered — you cannot slice or index them:
# s[0]  # TypeError: 'set' object is not subscriptable

# 3. set() is mutable; frozenset() is the immutable variant (can be hashed).
