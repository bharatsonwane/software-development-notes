nums = [4, 2, 8, 5, 1]

nums.extend([9, 3])
print("After extend:", nums)

nums.remove(8)  # Remove by value

nums.reverse()
print("After reverse:", nums)

nums.sort(reverse=True)
print("After sort desc:", nums)
  3. Task 3: Nested lists — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 3: Nested lists (2D Matrix operations)
matrix = [
    [1, 2, 3],
    [4, 5, 6],
    [7, 8, 9]
]

print("Access row 1, col 2:", matrix[1][2]) # 6

# Transpose matrix:
transposed = [[row[col] for row in matrix] for col in range(len(matrix[0]))]
print("Transposed:")
for r in transposed:
    print(" ", r)
  4. Task 4: List comprehension — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 4: List comprehension
# Squares of even numbers from 1 to 20:
evens_squared = [x**2 for x in range(1, 21) if x % 2 == 0]
print("Evens squared:", evens_squared)

# Flattening a 2D list with comprehension:
nested = [[1, 2], [3, 4], [5, 6]]
flat = [item for sublist in nested for item in sublist]
print("Flattened:", flat)

P.3. Challenge

  1. Mini project — Combine two topics from this file into one short script related to Lists. ▾ Solution 
    # Mini Project: Student Gradebook & Statistics Aggregator
students = [
    {"name": "Alice", "grades": [88, 92, 79]},
    {"name": "Bob", "grades": [75, 80, 85]},
    {"name": "Charlie", "grades": [95, 98, 92]},
    {"name": "Diana", "grades": [60, 65, 70]}
]

# Calculate averages using list comprehensions:
report = [
    {
        "name": s["name"],
        "avg": sum(s["grades"]) / len(s["grades"]),
        "highest": max(s["grades"])
    }
    for s in students
]

# Sort students by average descending:
report.sort(key=lambda item: item["avg"], reverse=True)

print(f"{'Student':<10} {'Average':<8} {'Highest':<8}")
print("-" * 28)
for r in report:
    print(f"{r['name']:<10} {r['avg']:<8.2f} {r['highest']:<8}")
  2. Edge cases — List 3 edge cases or mistakes beginners make; demonstrate at least one safely. ▾ Solution 
    # List Pitfalls & Edge Cases:
# 1. Pitfall: [[]] * 3 creates shallow copies pointing to SAME inner list:
bad_matrix = [[]] * 3
bad_matrix[0].append(1)
print("Bad matrix (all rows mutated!):", bad_matrix)  # [[1], [1], [1]]

# Fix: Use list comprehension for independent instances:
good_matrix = [[] for _ in range(3)]
good_matrix[0].append(1)
print("Good matrix (independent):", good_matrix)      # [[1], [], []]

# 2. remove() raises ValueError if element is not in list.
