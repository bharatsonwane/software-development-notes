    # Summary of Dict Properties, Creation, and Access:
# 1. Dictionaries store key-value mappings where keys must be unique and hashable (immutable).
# 2. Since Python 3.7+, dictionaries preserve insertion order.
# 3. Accessing a missing key via d[key] raises KeyError; d.get(key, default) safely returns default.

P.2. Core tasks

  1. Task 1: dict properties, creation, access — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 1: dict properties, creation, access
# Creation styles:
d1 = {"a": 1, "b": 2}
d2 = dict(name="Bob", age=30)
d3 = dict([("x", 10), ("y", 20)])

print("d1:", d1)
print("d2:", d2)
print("d3:", d3)

# Safe access with .get():
print("d2['name']:", d2["name"])
print("d2.get('salary', 'N/A'):", d2.get("salary", "N/A"))
  2. Task 2: Update, delete, and important methods — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 2: Update, delete, and important methods
profile = {"user": "charlie", "email": "charlie@test.com", "tier": "free"}
print("keys():", list(profile.keys()))
print("values():", list(profile.values()))
print("items():", list(profile.items()))

# setdefault: inserts key with default if not already present
profile.setdefault("tier", "premium") # tier already exists, unchanged
profile.setdefault("points", 100)     # inserted
print("Profile after setdefault:", profile)
  3. Task 3: Dict programs and comprehension — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 3: Dict programs and comprehension
# Character frequency counter:
text = "dictionary programming"
char_counts = {ch: text.count(ch) for ch in set(text) if ch != " "}
print("Character counts:", char_counts)

# Inverting a dictionary (values become keys):
original = {"apple": "fruit", "carrot": "vegetable", "banana": "fruit"}
# Grouping by category:
grouped = {}
for item, cat in original.items():
    grouped.setdefault(cat, []).append(item)
print("Grouped by category:", grouped)
  4. Task 4: Merging and nested collections — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 4: Merging and nested collections
# Merging dictionaries (Python 3.9+ '|' operator):
default_settings = {"theme": "light", "notifications": True, "timeout": 30}
user_settings = {"theme": "dark", "timeout": 60}

merged = default_settings | user_settings  # Right-side overrides
print("Merged settings:", merged)

# Nested dictionary traversal:
company = {
    "tech": {
        "lead": "Alice",
        "team": ["Bob", "Charlie"]
    }
}
print("Tech team members:", company["tech"]["team"])

P.3. Challenge

  1. Mini project — Combine two topics from this file into one short script related to Dictionaries. ▾ Solution 
    # Mini Project: In-Memory Product Catalog & Inventory Manager
catalog = {
    "SKU-001": {"name": "Wireless Mouse", "price": 29.99, "stock": 45},
    "SKU-002": {"name": "Mechanical Keyboard", "price": 99.50, "stock": 12},
    "SKU-003": {"name": "USB-C Hub", "price": 49.00, "stock": 0},
    "SKU-004": {"name": "Monitor Arm", "price": 79.99, "stock": 25}
}

def purchase_product(catalog, sku, qty):
    product = catalog.get(sku)
    if not product:
        return False, "Product not found"
    if product["stock"] < qty:
        return False, f"Insufficient stock ({product['stock']} available)"
    product["stock"] -= qty
    total_cost = product["price"] * qty
    return True, f"Purchased {qty}x {product['name']} for ${total_cost:.2f}"

success, msg = purchase_product(catalog, "SKU-002", 2)
print("Order status:", msg)
print("Updated stock for SKU-002:", catalog["SKU-002"]["stock"])
  2. Edge cases — List 3 edge cases or mistakes beginners make; demonstrate at least one safely. ▾ Solution 
    # Dict Pitfalls & Edge Cases:
# 1. Unhashable keys: lists cannot be dictionary keys:
# d = {[1, 2]: "data"} # TypeError: unhashable type: 'list'
# Use tuples instead: d = {(1, 2): "data"}

# 2. Modifying dict size during iteration raises RuntimeError:
# for k in d: del d[k] # RuntimeError: dictionary changed size during iteration
# Fix: iterate over list(d.keys())

# 3. Missing key without get() raises KeyError.
