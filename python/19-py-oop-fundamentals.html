    # Summary of Class, Object, and Reference Variable:
# 1. A class is a blueprint for creating objects, defining attributes and methods.
# 2. An object is a concrete instance of a class allocated in memory.
# 3. A reference variable points to the object in heap memory; multiple references can point to the same object.
  2. Create a tiny example related to self variable (complete story) and print the result. ▾ Solution 
    # The 'self' variable:
class Greeter:
    def set_name(self, name):
        # 'self' refers to the current instance calling the method
        self.name = name
    def say_hello(self):
        print(f"Hello, my name is {self.name} (id={id(self)})")

g = Greeter()
g.set_name("Bharat")
g.say_hello()

class Car:
    pass

# Creating multiple objects:
car1 = Car()
car2 = Car()
car3 = car1  # Aliasing reference

print("car1 address:", hex(id(car1)))
print("car2 address:", hex(id(car2)))
print("car1 is car2:", car1 is car2)  # False (independent objects)
print("car1 is car3:", car1 is car3)  # True (same reference)
  2. Task 2: self variable (complete story) — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 2: self variable (complete story)
class Device:
    def __init__(self, brand, model):
        # self binds attributes directly to the specific instance
        self.brand = brand
        self.model = model
    
    def print_specs(self):
        print(f"Device: {self.brand} {self.model}")

d1 = Device("Apple", "MacBook Pro")
d2 = Device("Dell", "XPS 15")

# Behind the scenes, Device.print_specs(d1) is invoked:
d1.print_specs()
Device.print_specs(d2)

  4. Task 4: Instance variables and methods — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 4: Instance variables and methods
class BankAccount:
    bank_name = "Global Tech Bank"  # Class / Static variable
    
    def __init__(self, owner, balance=0.0):
        self.owner = owner          # Instance variable
        self.balance = balance      # Instance variable
    
    def deposit(self, amount):      # Instance method
        if amount > 0:
            self.balance += amount
            print(f"Deposited ${amount:.2f}. New Balance: ${self.balance:.2f}")
    
    def withdraw(self, amount):     # Instance method
        if 0 < amount <= self.balance:
            self.balance -= amount
            print(f"Withdrew ${amount:.2f}. Remaining: ${self.balance:.2f}")
        else:
            print("Insufficient funds or invalid amount.")

acc = BankAccount("Bharat", 500.0)
acc.deposit(200.0)
acc.withdraw(150.0)
print(f"Account belongs to {acc.owner} at {BankAccount.bank_name}")

P.3. Challenge

  1. Mini project — Combine two topics from this file into one short script related to OOP Fundamentals. ▾ Solution 
    # Mini Project: Employee Management System with Class & Static Methods
class Employee:
    company = "InnovateCorp"
    employee_count = 0
    
    def __init__(self, emp_id, name, base_salary):
        self.emp_id = emp_id
        self.name = name
        self.base_salary = base_salary
        Employee.employee_count += 1
    
    @classmethod
    def get_company_stats(cls):
        return f"{cls.company} employs {cls.employee_count} active team members."
    
    @staticmethod
    def calculate_bonus(salary, rating):
        # Pure utility calculation
        multiplier = 0.20 if rating >= 4.5 else 0.10
        return round(salary * multiplier, 2)

e1 = Employee("EMP-1", "Sarah", 95000)
e2 = Employee("EMP-2", "David", 82000)

print(Employee.get_company_stats())
print(f"{e1.name}'s Bonus: ${Employee.calculate_bonus(e1.base_salary, 4.8):,.2f}")
  2. Edge cases — List 3 edge cases or mistakes beginners make; demonstrate at least one safely. ▾ Solution 
    # OOP Fundamentals Pitfalls:
# 1. Modifying class variable via self creates a shadowing instance variable:
class Config:
    timeout = 30

c = Config()
c.timeout = 60  # Creates c.__dict__['timeout'] = 60; Config.timeout remains 30!
print("c.timeout:", c.timeout)           # 60
print("Config.timeout:", Config.timeout) # 30

# 2. Omitting 'self' as first parameter in instance methods causes TypeError on invocation.
