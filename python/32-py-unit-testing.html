P.2. Core tasks

  1. Task 1: unittest introduction — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 1: unittest.TestCase demonstration
import unittest

def add(a, b):
    return a + b

def divide(a, b):
    if b == 0:
        raise ValueError("Denominator cannot be zero")
    return a / b

class TestMathOperations(unittest.TestCase):
    def test_add(self):
        self.assertEqual(add(3, 4), 7)
        self.assertEqual(add(-1, 1), 0)
    
    def test_divide_success(self):
        self.assertEqual(divide(10, 2), 5.0)
    
    def test_divide_zero(self):
        with self.assertRaises(ValueError):
            divide(10, 0)

# Programmatic test suite execution:
suite = unittest.TestLoader().loadTestsFromTestCase(TestMathOperations)
runner = unittest.TextTestRunner(verbosity=2)
runner.run(suite)

  3. Task 3: Limitations of unittest; intro to pytest — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 3: Limitations of unittest vs Pytest
# Comparison:
# unittest: Requires classes, self.assert*, boilerplate test runners.
# pytest: Simple assert statements, parameterization, powerful fixtures, no boilerplate.

def is_prime(n):
    return n > 1 and all(n % i != 0 for i in range(2, int(n**0.5) + 1))

# Pytest style test function (pure function with assert):
def test_is_prime():
    assert is_prime(2) == True
    assert is_prime(7) == True
    assert is_prime(4) == False

test_is_prime()
print("Pytest test passed cleanly without class boilerplate.")
  4. Task 4: setUp / tearDown / class-level fixtures in pytest — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 4: setUp / tearDown / class-level fixtures
import unittest

class TestDatabaseConnection(unittest.TestCase):
    @classmethod
    def setUpClass(cls):
        print("\n[setUpClass] Connecting to test database instance once...")
    
    def setUp(self):
        print("[setUp] Initializing transaction for test method")
    
    def test_query_one(self):
        self.assertTrue(True)
    
    def tearDown(self):
        print("[tearDown] Rolling back test transaction")
    
    @classmethod
    def tearDownClass(cls):
        print("[tearDownClass] Closing database connection pool.")

suite = unittest.TestLoader().loadTestsFromTestCase(TestDatabaseConnection)
unittest.TextTestRunner(verbosity=0).run(suite)

  1. Mini project — Combine two topics from this file into one short script related to Unit Testing. ▾ Solution 
    # Mini Project: Shopping Cart Test Suite with Mock Payment Gateway
import unittest
from unittest.mock import MagicMock

class ShoppingCart:
    def __init__(self, payment_gateway):
        self.gateway = payment_gateway
        self.items = []
    
    def add_item(self, name, price):
        self.items.append(price)
    
    def total(self):
        return sum(self.items)
    
    def checkout(self):
        return self.gateway.charge(self.total())

class TestShoppingCart(unittest.TestCase):
    def test_checkout_success(self):
        mock_gateway = MagicMock()
        mock_gateway.charge.return_value = True
        
        cart = ShoppingCart(mock_gateway)
        cart.add_item("Book", 25.0)
        cart.add_item("Pen", 5.0)
        
        self.assertEqual(cart.total(), 30.0)
        result = cart.checkout()
        
        self.assertTrue(result)
        mock_gateway.charge.assert_called_once_with(30.0)

suite = unittest.TestLoader().loadTestsFromTestCase(TestShoppingCart)
unittest.TextTestRunner(verbosity=0).run(suite)
print("Shopping cart test passed with mocked payment gateway.")
