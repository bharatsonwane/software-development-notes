P.2. Core tasks

  1. Task 1: Defining functions, parameters, return — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 1: Defining functions, parameters, return
def calculate_metrics(numbers):
    """Return count, sum, and average of a list of numbers."""
    if not numbers:
        return 0, 0, 0.0
    total = sum(numbers)
    count = len(numbers)
    avg = total / count
    return count, total, avg

c, s, a = calculate_metrics([10, 20, 30, 40])
print(f"Count: {c}, Sum: {s}, Average: {a:.2f}")
  2. Task 2: Positional, keyword, default, variable-length args — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 2: Positional, keyword, default, variable-length args (*args, **kwargs)
def build_query(table, *columns, **filters):
    cols_str = ", ".join(columns) if columns else "*"
    conditions = " AND ".join(f"{k}='{v}'" for k, v in filters.items())
    query = f"SELECT {cols_str} FROM {table}"
    if conditions:
        query += f" WHERE {conditions}"
    return query

print(build_query("users", "id", "name", "email", status="active", role="admin"))
print(build_query("orders", status="shipped"))
  3. Task 3: Local and global variables; global keyword — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 3: Local and global variables; global keyword
counter = 0  # Global variable

def increment():
    global counter  # Declare intent to modify global variable
    counter += 1

def read_only():
    # Reading global without modification does not require global keyword
    print("Reading global counter:", counter)

increment()
increment()
read_only()
print("Final counter value:", counter)
  4. Task 4: Recursive functions — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 4: Recursive functions
def factorial(n):
    """Compute factorial recursively with base case n <= 1."""
    if n < 0:
        raise ValueError("Factorial not defined for negative numbers")
    if n <= 1:
        return 1
    return n * factorial(n - 1)

print("5! =", factorial(5))  # 120
print("7! =", factorial(7))  # 5040

    # Mini Project: Flexible Math Expression Pipeline
def math_pipeline(*funcs):
    """Return a function that chains multiple transformations."""
    def pipeline(val):
        result = val
        for f in funcs:
            result = f(result)
        return result
    return pipeline

double = lambda x: x * 2
add_ten = lambda x: x + 10
square = lambda x: x ** 2

compute = math_pipeline(double, add_ten, square)
# (5 * 2 = 10) -> (10 + 10 = 20) -> (20 ** 2 = 400)
print("Pipeline result for 5:", compute(5))
  2. Edge cases — List 3 edge cases or mistakes beginners make; demonstrate at least one safely. ▾ Solution 
    # Function Pitfalls & Edge Cases:
# 1. Dangerous Default Argument Trap (mutable default):
# def append_to(item, lst=[]): lst.append(item); return lst
# Fix: Use None as default:
def safe_append(item, lst=None):
    if lst is None:
        lst = []
    lst.append(item)
    return lst

print("Call 1:", safe_append(1))  # [1]
print("Call 2:", safe_append(2))  # [2] (independent!)
