    # Summary of HAS-A Relationship (Composition / Aggregation):
# 1. HAS-A models composition where a class contains references to instances of other classes.
# 2. Promotes loose coupling and reuse without creating deep, brittle inheritance hierarchies.
# 3. Principle: 'Favor composition over inheritance' whenever possible.
  2. Create a tiny example related to IS-A relationship (inheritance) and print the result. ▾ Solution 
    # IS-A Relationship (Inheritance):
class Animal:
    def breathe(self):
        return "Breathing oxygen"

class Dog(Animal):  # Dog IS-A Animal
    def bark(self):
        return "Woof!"

d = Dog()
print("Dog breathing:", d.breathe())
print("Dog barking:", d.bark())
print("isinstance(d, Animal):", isinstance(d, Animal))

P.2. Core tasks

  1. Task 1: HAS-A relationship (composition / aggregation) — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 1: HAS-A relationship (composition / aggregation)
class Engine:
    def start(self):
        return "V8 Engine started rumbling."

class Car:
    def __init__(self, brand):
        self.brand = brand
        self.engine = Engine()  # Car HAS-A Engine
    
    def drive(self):
        status = self.engine.start()
        return f"{self.brand} is moving. {status}"

c = Car("Mustang")
print(c.drive())
  2. Task 2: IS-A relationship (inheritance) — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 2: IS-A relationship with super() constructor chaining
class Person:
    def __init__(self, name, age):
        self.name = name
        self.age = age
    
    def get_info(self):
        return f"{self.name}, Age {self.age}"

class SoftwareEngineer(Person):
    def __init__(self, name, age, tech_stack):
        super().__init__(name, age)  # Call parent constructor
        self.tech_stack = tech_stack
    
    def get_info(self):
        base = super().get_info()
        return f"{base} | Stack: {', '.join(self.tech_stack)}"

eng = SoftwareEngineer("Bharat", 30, ["Python", "TypeScript", "React"])
print(eng.get_info())
  3. Task 3: Types of inheritance — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 3: Types of inheritance
# 1. Single: Child(Parent)
# 2. Multilevel: C(B), B(A)
# 3. Hierarchical: B(A), C(A)
# 4. Multiple: Child(ParentA, ParentB)

class Swimmer:
    def swim(self):
        return "Swimming in water."

class Flyer:
    def fly(self):
        return "Flying in the air."

class Duck(Swimmer, Flyer):  # Multiple inheritance
    pass

donald = Duck()
print(donald.swim())
print(donald.fly())
  4. Task 4: Method Resolution Order (MRO) — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 4: Method Resolution Order (MRO)
class A:
    def ping(self): print("A.ping")

class B(A):
    def ping(self): print("B.ping")

class C(A):
    def ping(self): print("C.ping")

class D(B, C):
    pass

d = D()
d.ping()  # Resolves to B.ping (first in MRO)
print("MRO for class D:")
for cls in D.mro():
    print(" ", cls.__name__)

    # Mini Project: UI Component Hierarchy with Composite Rendering
class UIComponent:
    def __init__(self, component_id):
        self.component_id = component_id
    
    def render(self):
        raise NotImplementedError

class Button(UIComponent):
    def __init__(self, component_id, label):
        super().__init__(component_id)
        self.label = label
    def render(self):
        return f"<button id='{self.component_id}'>{self.label}</button>"

class Panel(UIComponent):
    def __init__(self, component_id):
        super().__init__(component_id)
        self.children = []  # Composition
    
    def add(self, child):
        self.children.append(child)
    
    def render(self):
        inner = "\n  ".join(child.render() for child in self.children)
        return f"<div id='{self.component_id}'>\n  {inner}\n</div>"

panel = Panel("main-container")
panel.add(Button("btn-submit", "Submit"))
panel.add(Button("btn-cancel", "Cancel"))
print("Rendered Tree:\n" + panel.render())
