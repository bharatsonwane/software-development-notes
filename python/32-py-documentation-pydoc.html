def calculate_hypotenuse(a: float, b: float) -> float:
    """Calculate the length of the hypotenuse of a right-angled triangle.
    
    Args:
        a (float): Length of the first perpendicular side.
        b (float): Length of the second perpendicular side.

        float: Length of the hypotenuse (sqrt(a^2 + b^2)).

print("Function docstring:\n" + calculate_hypotenuse.__doc__)

  2. Task 2: Docstrings and generating docs — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 2: Docstrings and generating docs (Class and Methods)
class MathService:
    """High-performance mathematical calculations engine.
    
    Attributes:
        precision (int): Number of decimal places to round results.
    """
    
    def __init__(self, precision: int = 4):
        """Initialize the math engine with a precision level."""
        self.precision = precision
    
    def power(self, base: float, exponent: float) -> float:
        """Raise base to the given exponent rounded to configured precision."""
        return round(base ** exponent, self.precision)

import pydoc
doc_text = pydoc.render_doc(MathService, "Help on %s")
print("Rendered Pydoc Output (excerpt):")
print("\n".join(doc_text.splitlines()[:12]))
  3. Cheat sheet — Comment 5 key facts about Documentation (Pydoc). ▾ Solution 
    # Cheat sheet — 5 Key Facts about Documentation (Pydoc):
# 1. Triple-quoted docstrings placed immediately after def/class become the __doc__ attribute.
# 2. PEP 257 standardizes docstrings into one-line and multi-line formats.
# 3. Leading and trailing quotes should be on their own line for multi-line docstrings.
# 4. Standard formats include Google Style, Sphinx/reStructuredText, and NumPy style.
# 5. Type hints in signatures are automatically parsed and displayed by modern pydoc and IDEs.

  1. Mini project — Combine two topics from this file into one short script related to Documentation (Pydoc). ▾ Solution 
    # Mini Project: Automated Docstring Coverage Inspector
import inspect

def inspect_docstring_coverage(module_or_class):
    members = inspect.getmembers(module_or_class, predicate=inspect.isfunction)
    total = len(members)
    documented = sum(1 for name, fn in members if fn.__doc__ and fn.__doc__.strip())
    
    print(f"Documentation Audit for {module_or_class.__name__}:")
    print(f"  Total functions: {total}")
    print(f"  Documented:      {documented}")
    coverage = (documented / total * 100) if total else 100.0
    print(f"  Coverage:        {coverage:.1f}%")
    
    for name, fn in members:
        status = "[DOC OK]" if fn.__doc__ else "[MISSING]"
        print(f"  {status:<10} {name}()")

class SampleService:
    def method_one(self): """Does first operation.""" pass
    def method_two(self): pass
    def method_three(self): """Does third operation.""" pass

inspect_docstring_coverage(SampleService)
