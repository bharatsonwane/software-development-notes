  2. Task 2: import forms, naming conflicts, reloading — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 2: Import forms, naming conflicts, reloading
import importlib
import math

# Direct alias:
from datetime import datetime as dt
now = dt.now()
print("Current timestamp:", now.strftime("%Y-%m-%d %H:%M:%S"))

# Reloading a module dynamically at runtime:
importlib.reload(math)
print("Module math reloaded successfully.")
  3. Task 3: dir(), help(), __doc__, __file__, __name__ — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 3: dir(), help(), __doc__, __file__, __name__
import math

print("__name__ of current script:", __name__)
print("math.__file__ location:", getattr(math, '__file__', 'built-in C module'))
print("math.__doc__ excerpt:", math.__doc__[:60] if math.__doc__ else "N/A")

# dir() introspects all symbols exported by a module:
math_symbols = [s for s in dir(math) if not s.startswith("_")]
print(f"math exports {len(math_symbols)} symbols. Samples:", math_symbols[:8])
  4. Task 4: random module and applications — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 4: Random module and applications
import random

# Reproducible randomness with seed:
random.seed(42)

print("random() float in [0.0, 1.0):", random.random())
print("randint(1, 100):", random.randint(1, 100))
print("randrange(0, 100, 10):", random.randrange(0, 100, 10))

colors = ["red", "blue", "green", "gold", "purple"]
print("choice():", random.choice(colors))
print("sample() 3 unique:", random.sample(colors, 3))

deck = list(range(1, 11))
random.shuffle(deck)
print("shuffled deck:", deck)

  1. Mini project — Combine two topics from this file into one short script related to Modules. ▾ Solution 
    # Mini Project: Dice Simulator & Probability Estimator
import random

def estimate_dice_probabilities(trials=10000):
    sums_count = {}
    for _ in range(trials):
        dice1 = random.randint(1, 6)
        dice2 = random.randint(1, 6)
        total = dice1 + dice2
        sums_count[total] = sums_count.get(total, 0) + 1
    
    print(f"Simulation Results over {trials:,} rolls of two dice:")
    print(f"{'Sum':<5} {'Count':<8} {'Estimated %':<12}")
    print("-" * 28)
    for total in sorted(sums_count.keys()):
        pct = (sums_count[total] / trials) * 100
        print(f"{total:<5} {sums_count[total]:<8} {pct:>6.2f}%")

estimate_dice_probabilities(5000)
