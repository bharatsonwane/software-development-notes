    # Summary of Object Serialization with Pickle:
# 1. Serialization (pickling) converts in-memory Python object hierarchies into byte streams.
# 2. Deserialization (unpickling) reconstructs the Python object from the byte stream.
# 3. CAUTION: 'pickle' is Python-specific and NOT secure against malicious data. Never unpickle untrusted bytes.

import json

data = {"project": "Python Notes", "version": 3.12, "active": True, "tags": ["backend", "docs"]}

# Serialization to string:
json_str = json.dumps(data, indent=2)

# Deserialization from string:
parsed_data = json.loads(json_str)
print("Deserialized title:", parsed_data["project"])

  1. Task 1: Object serialization with pickle — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 1: Object serialization with pickle
import pickle

class Player:
    def __init__(self, username, level, score):
        self.username = username
        self.level = level
        self.score = score

player = Player("GamerOne", 15, 9420)

# Serialize to bytes:
pickled_bytes = pickle.dumps(player)
print(f"Pickled bytes length: {len(pickled_bytes)}")

# Deserialize from bytes:
restored_player = pickle.loads(pickled_bytes)
print(f"Restored Player: {restored_player.username}, Level: {restored_player.level}, Score: {restored_player.score}")

import json

payload = {

  3. Task 3: jsonpickle and API JSON examples — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 3: Custom JSON encoding for Python objects
import json
from datetime import datetime

class Event:
    def __init__(self, name, timestamp):
        self.name = name
        self.timestamp = timestamp

class EventEncoder(json.JSONEncoder):
    def default(self, obj):
        if isinstance(obj, Event):
            return {"event_name": obj.name, "timestamp": obj.timestamp.isoformat()}
        return super().default(obj)

event = Event("DEPLOYMENT", datetime(2026, 9, 13, 12, 0, 0))
encoded_event = json.dumps(event, cls=EventEncoder)
print("Custom JSON encoded object:", encoded_event)

  1. Mini project — Combine two topics from this file into one short script related to Serialization (pickle, JSON, YAML). ▾ Solution 
    # Mini Project: Cache Serializer Supporting Multiple Formats
import json
import pickle

class SerializerCache:
    @staticmethod
    def serialize(data, format="json"):
        if format == "json":
            return json.dumps(data).encode("utf-8")
        elif format == "pickle":
            return pickle.dumps(data)
        raise ValueError(f"Unsupported format: {format}")
    
    @staticmethod
    def deserialize(raw_bytes, format="json"):
        if format == "json":
            return json.loads(raw_bytes.decode("utf-8"))
        elif format == "pickle":
            return pickle.loads(raw_bytes)
        raise ValueError(f"Unsupported format: {format}")

sample = {"session_id": "xyz987", "user_id": 402, "permissions": ["read", "write"]}

b_json = SerializerCache.serialize(sample, "json")
b_pickle = SerializerCache.serialize(sample, "pickle")

print("JSON serialized size:", len(b_json), "bytes")
print("Pickle serialized size:", len(b_pickle), "bytes")
print("Verified deserialized match:", SerializerCache.deserialize(b_json, "json") == sample)
