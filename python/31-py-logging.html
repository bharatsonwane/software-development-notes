# 1. Python's 'logging' standard library tracks events during application runtime.
# 2. Five standard severity levels: DEBUG (10) < INFO (20) < WARNING (30) < ERROR (40) < CRITICAL (50).
# 3. Default logging level is WARNING; events below WARNING are ignored unless configured via basicConfig.

    1 / 0

import logging

logger = logging.getLogger("demo_logger")
logger.setLevel(logging.DEBUG)

# Console handler:
ch = logging.StreamHandler()
ch.setLevel(logging.DEBUG)
formatter = logging.Formatter("%(levelname)s: %(message)s")
ch.setFormatter(formatter)
logger.addHandler(ch)

logger.debug("Detailed diagnostic information (DEBUG)")
logger.info("Normal lifecycle milestone (INFO)")
logger.warning("Unusual condition that requires attention (WARNING)")
logger.error("Operation failed (ERROR)")

  2. Task 2: Formatting messages; logging exceptions — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 2: Formatting messages and logging exceptions
import logging

formatter = logging.Formatter(
    fmt="%(asctime)s | %(name)s | %(levelname)-8s | %(filename)s:%(lineno)d | %(message)s",
    datefmt="%Y-%m-%d %H:%M:%S"
)
record = logging.LogRecord(
    name="payment_service", level=logging.ERROR, pathname="payment.py",
    lineno=42, msg="Card declined: insufficient funds", args=(), exc_info=None
)
print("Formatted Log Line:")
print(formatter.format(record))
  3. Task 3: Custom logger patterns — Write a small working demo (10–20 lines) covering the idea; print outputs. ▾ Solution 
    # Task 3: Custom logger patterns (File and Console handlers)
import logging
import os

app_logger = logging.getLogger("app.service")
app_logger.setLevel(logging.INFO)

log_filename = "app_activity.log"
file_handler = logging.FileHandler(log_filename, mode="w")
file_handler.setFormatter(logging.Formatter("[%(asctime)s] %(levelname)s - %(message)s"))

app_logger.addHandler(file_handler)
app_logger.info("Service started successfully.")
app_logger.info("Processing queue batch #104.")

file_handler.close()
with open(log_filename) as f:
    print("Log file output:\n" + f.read())
os.remove(log_filename)

  1. Mini project — Combine two topics from this file into one short script related to Logging. ▾ Solution 
    # Mini Project: Robust Request Logging Middleware Simulator
import logging
import time

logger = logging.getLogger("api_middleware")
logger.setLevel(logging.INFO)
handler = logging.StreamHandler()
handler.setFormatter(logging.Formatter("%(asctime)s [%(levelname)s] %(message)s"))
logger.addHandler(handler)

def log_api_request(endpoint, handler_fn):
    start = time.perf_counter()
    logger.info(f"Incoming Request -> {endpoint}")
    try:
        response = handler_fn()
        elapsed = (time.perf_counter() - start) * 1000
        logger.info(f"Completed {endpoint} in {elapsed:.2f}ms with status 200 OK")
        return response
    except Exception as err:
        logger.error(f"Failed {endpoint}: {err}", exc_info=True)
        return {"error": str(err), "status": 500}

# Simulate successful request:
log_api_request("/api/v1/health", lambda: {"status": "healthy"})
